#include<iostream>
#include<vector>
#include<string>
using namespace std;

class BigInt{
	vector<int> digit;
	bool sign=true;
	vector<int> add(const vector<int> &a,const vector<int> &b){
		vector<int> res;
		int rec=0;
		for(size_t i=0;i<a.size()||i<b.size()||rec;i++){
			if(i<a.size()){
				rec+=a[i];
			}
			if(i<b.size()){
				rec+=b[i];
			}
			res.push_back(rec%10);
			rec/=10;
		}
		remove_zero(res);
		return res;
	}
	vector<int> sub(const vector<int> &a,const vector<int> &b){
		vector<int> res;
		int bow=0;
		for(size_t i=0;i<a.size();i++){
			int tmp=a[i];
			if(i<b.size()){
				tmp-=b[i];
			}
			tmp-=bow;
			bow=(tmp<0)?1:0;
			res.push_back((tmp+10)%10);
		}
		remove_zero(res);
		return res;
	}
	vector<int> mul(const vector<int> &a,const vector<int> &b){
		vector<int> res(a.size()+b.size(),0);
		for(size_t i=0;i<a.size();i++){
			for(size_t j=0;j<b.size();j++){
				res[i+j]+=a[i]*b[j];
				res[i+j+1]+=res[i+j]/10;
				res[i+j]%=10;
			}
		}
		remove_zero(res);
		return res;
	}
	bool bigger(const vector<int> &a,const vector<int> &b){
		if(a.size()!=b.size()){
			return a.size()>b.size();
		}
		for(int i=a.size()-1;i>=0;i--){
			if(a[i]!=b[i]){
				return a[i]>b[i];
			}
		}
		return true;
	}
	void remove_zero(vector<int> &arr){
		while(arr.size() > 1 && arr.back() == 0){
			arr.pop_back();
		}
	}
	void check_zero(){
		if(this->digit.size()==1&&this->digit[0]==0){
			this->sign=true;
		}
	}
public:
	BigInt()=default;
	BigInt(const char *arr):BigInt(string(arr)){}
	BigInt(const string &str){
		int flag=0;
		if(str[0]=='-'){
			sign=false;
			flag=1;
		}
		for(int i=str.size()-1;i>=flag;i--){
			digit.push_back(str[i]-'0');
		}
	}
	BigInt(int num){
		if(num<0){
			sign=false;
			num=-num;
		}
		do{
			digit.push_back(num%10);
			num/=10;
		}while(num);
	}
	friend ostream& operator<<(ostream& out, const BigInt& a) {
		if (!a.sign) {
			out << '-';
		}
		for(int i=a.digit.size()-1;i>=0;i--){
			out<<a.digit[i];
		}
		return out;
	}
	friend istream& operator>>(istream& in, BigInt& a) {
		string str;
		in >> str;
		a = BigInt(str);
		return in;
	}
	BigInt operator-() const {
		BigInt res=*this;
		if(res.digit.size()>1||res.digit[0]!=0){{
			res.sign=!res.sign;
		}}
		return res;
	}
	void operator+=(const BigInt &b){
		if(this->sign==b.sign){
			this->digit=add(this->digit,b.digit);
		}else{
			if(bigger(this->digit,b.digit)){
				this->digit=sub(this->digit,b.digit);
			}else{
				this->digit=sub(b.digit,this->digit);
				this->sign=b.sign;
			}
		}
		check_zero();
	}
	void operator-=(const BigInt &b){
		*this+=(-b);
	}
	void operator*=(const BigInt &b){
		this->digit=mul(this->digit,b.digit);
		this->sign=this->sign==b.sign;
		check_zero();
	}
	void operator/=(const BigInt &b){
		
	}
	BigInt operator+(const BigInt &b){
		BigInt res=*this;
		res+=b;
		return res;
	}
	BigInt operator-(const BigInt &b){
		BigInt res=*this;
		res-=b;
		return res;
	}
	BigInt operator*(const BigInt &b){
		BigInt res=*this;
		res*=b;
		return res;
	}
};
