#include<iostream>

#include<string>
using namespace std;

class BigInt {
	vector<int> digits;
	bool sign;
	void remove_zero() {
		while (digits.size() > 1 && digits.back() == 0) {
			digits.pop_back();
		}
		if (digits.size() == 1 && digits.back() == 0) {
			sign = true;
		}
	}
	bool bigger_abs(const vector<int> &a, const vector<int> &b) {
		if (a.size() != b.size()) {
			return a.size() > b.size();
		}
		for(int i=a.size()-1;i>=0;i--){
			if (a[i] != b[i]) {
				return a[i] > b[i];
			}
		}
		return true;
	}
	pair<BigInt,BigInt> divide(const BigInt &b){
		BigInt dividend=0,devisor=b,reminder=0,res=0;
		devisor.sign=true;
		for(int i=this->digits.size()-1;i>=0;i--){
			int cur=this->digits[i];
			dividend=reminder*10+cur;
			reminder=dividend;
			int q=0;
			while(reminder>=devisor){
				reminder-=devisor;
				q++;
			}
			res=res*10+q;
		}
		return pair<BigInt,BigInt>{res,reminder};
	}
public:
	BigInt() = default;
	BigInt(const char* arr): BigInt(string(arr)) {}
	BigInt(const string& str): sign(true) {
		if (str.empty()) {
			digits.push_back(0);
		}
		int start = 0;
		if (str[0] == '-') {
			sign = false;
			start = 1;
		} else if (str[0] == '+') {
			start = 1;
		}
		for (int i = str.size() - 1; i >= start; i--) {
			if (str[i] < '0' || str[i] > '9') {
				throw invalid_argument("错误");
			}
			digits.push_back(str[i] - '0');
		}
		remove_zero();
	}
	BigInt(int num): sign(num >= 0) {
		if (num == 0) {
			digits.push_back(0);
		}
		if (num < 0) {
			num = -num;
		}
		while (num) {
			digits.push_back(num % 10);
			num /= 10;
		}
	}
	friend istream& operator>>(istream& is, BigInt& a) {

		is >> str;
		a = BigInt(str);

		if (!a.sign) {
			os << "-";
		}
		for (int i = a.digits.size() - 1; i >= 0; i--) {
			os << a.digits[i];
		}
		return os;
	}
	BigInt operator-() {
		BigInt res = *this;
		if (res.digits.size() > 1 || res.digits[0] != 0) {
			res.sign = !res.sign;
		}
		return res;
	}
	BigInt& operator+=(const BigInt &b) {
		if (this->sign != b.sign) {
			BigInt tmp = b;
			tmp.sign = !tmp.sign;
			return *this -= tmp;
		}
		vector<int> res;
		int rec = 0;
		for (size_t i = 0; i < this->digits.size() || i < b.digits.size() || rec; i++) {
			if (i < this->digits.size()) {
				rec += this->digits[i];
			}
			if (i < b.digits.size()) {
				rec += b.digits[i];
			}
			res.push_back(rec % 10);
			rec /= 10;
		}
		this->digits = res;
		this->remove_zero();
		return *this;
	}
	BigInt& operator-=(const BigInt &b) {
		if (this->sign != b.sign) {
			BigInt tmp = b;
			tmp.sign = !tmp.sign;
			return *this += tmp;
		}
		bool is_big = bigger_abs(this->digits, b.digits);
		if (!is_big) {
			this->sign = !this->sign;
		}
		vector<int> s = is_big ? b.digits : this->digits;
		vector<int> h = is_big ? this->digits : b.digits;
		vector<int> res;
		int bow = 0;
		for (size_t i = 0; i < h.size(); i++) {
			int tmp = h[i] - bow;
			if (i < s.size()) {
				tmp -= s[i];
			}
			res.push_back((tmp + 10) % 10);
			bow = (tmp < 0) ? 1 : 0;
		}
		this->digits = res;
		this->remove_zero();
		return *this;
	}
	BigInt& operator*=(const BigInt &b) {
		vector<int> res(this->digits.size() + b.digits.size());
		for (size_t i = 0; i < this->digits.size(); i++) {
			for (size_t j = 0; j < b.digits.size(); j++) {
				res[i + j] += this->digits[i] * b.digits[j];
				res[i + j + 1] += res[i + j] / 10;
				res[i + j] %= 10;
			}
		}
		this->digits = res;
		this->sign = (this->sign == b.sign);
		this->remove_zero();
		return *this;
	}

			throw invalid_argument("除数是0");
		}
		pair<BigInt,BigInt> res=this->divide(b);

		this->remove_zero();

	}

		if (this->sign != b.sign || this->digits.size() != b.digits.size()) {
			return false;
		}
		for (size_t i = 0; i < this->digits.size(); i++) {
			if (this->digits[i] != b.digits[i]) {
				return false;

	}

			if(this->digits[i]!=b.digits[i]){
				return (this->sign == false) ^ (this->digits[i] > b.digits[i]);
			}
		}
		return false;
	}
	bool operator>=(const BigInt &b){
		return !(*this<b);
	}
	bool operator<(const BigInt &b){
		if (this->sign != b.sign) {
			return this->sign == false;
		}
		if (this->digits.size() != b.digits.size()) {
			return (this->sign == false) ^ (this->digits.size() < b.digits.size());
		}
		for(int i=this->digits.size()-1;i>=0;i--){
			if(this->digits[i]!=b.digits[i]){
				return (this->sign == false) ^ (this->digits[i] < b.digits[i]);
			}
		}
		return false;
	}

		return !(*this>b);

		return *this+=BigInt(1);

		BigInt tmp=*this;
		++(*this);
		return tmp;

		return *this-=BigInt(1);
	}

		BigInt tmp=*this;
		--(*this);
		return tmp;
	}
