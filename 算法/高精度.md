#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

// 将字符串转换为逆序数组（低位在前）
vector<int> str_to_vec(const string &s) {
	vector<int> res;
	for (int i = s.size() - 1; i >= 0; i--) {
		res.push_back(s[i] - '0');
	}
	return res;
}

// 将逆序数组转换为字符串
string vec_to_str(const vector<int> &v) {
	string res;
	for (int i = v.size() - 1; i >= 0; i--) {
		res += to_string(v[i]);
	}
	return res.empty() ? "0" : res;
}

vector<int> add(vector<int> &a, vector<int> &b) {
	vector<int> res;
	int carry = 0;
	for (size_t i = 0; i < a.size() || i < b.size() || carry; i++) {
		int sum = carry;
		if (i < a.size()) sum += a[i];
		if (i < b.size()) sum += b[i];
		carry = sum / 10;
		res.push_back(sum % 10);
	}
	return res;
}

vector<int> sub(vector<int> &a,vector<int> &b){
	vector<int> res;
	int borrow = 0;
	for (size_t i = 0; i < a.size(); i++) {
		int sub = a[i] - borrow;
		if (i < b.size()) sub -= b[i];
		borrow = (sub < 0) ? 1 : 0;
		res.push_back((sub + 10) % 10);
	}
	while (res.size() > 1 && res.back() == 0) res.pop_back();
	return res;
}

vector<int> mul(vector<int> &a, vector<int> &b) {
	vector<int> res(a.size() + b.size(), 0);
	for (size_t i = 0; i < a.size(); i++) {
		int carry = 0;
		for (size_t j = 0; j < b.size() || carry; j++) {
			int product = res[i + j] + a[i] * (j < b.size() ? b[j] : 0) + carry;
			res[i + j] = product % 10;
			carry = product / 10;
		}
	}
	while (res.size() > 1 && res.back() == 0) res.pop_back();
	return res;
}

vector<int> mul1(vector<int> &a,int b){
	vector<int> res;
	int carry=0;
	for(size_t i=0;i<a.size() || carry;i++)
	{
		if(i<a.size()) {
			carry+=a[i]*b;
		}
		res.push_back(carry%10);
		carry/=10;    
	}
	return res;
}

pair<vector<int>, int> divide(const vector<int> &a, int divisor) {
	vector<int> res;
	int remainder = 0;
	for (size_t i = a.size() - 1; i >= 0; i--) {
		remainder = remainder * 10 + a[i];
		res.push_back(remainder / divisor);
		remainder %= divisor;
	}
	reverse(res.begin(), res.end());
	while (res.size() > 1 && res.back() == 0) res.pop_back();
	return {res, remainder};
}

int main() {

	auto result = mul1(a, 2);

	return 0;
}

struct BigInt{
	vector<int> num;
	BigInt(int x){
		while(x!=0){
			num.push_back(x%10);
			x/=10;
		}
	}
	BigInt(const string &s) {
		for(int i=s.size()-1; i>=0; i--){
			num.push_back(s[i]-'0');
		}
	}
	void operator+=(int x){
		vector<int> res;
		int carry=0,n=num.size();
		for(int i=0;i<n||x||carry;i++){
			int sum=carry;
			if(i<n){
				sum+=num[i];
			}
			sum+=x%10;
			res.push_back(sum%10);
			x/=10;
			carry=sum/10;
		}
		num=res;
		return;
	}
	void operator-=(int x){
		vector<int> res;
		int borrow = 0,n=num.size();
		for (int i = 0; i < n; i++) {
			int sub = num[i] - borrow;
			sub -= x%10;
			res.push_back((sub + 10) % 10);
			borrow = (sub < 0) ? 1 : 0;
			x/=10;
		}
		while (res.size() > 1 && res.back() == 0) res.pop_back();
		num=res;
		return;
	}
	void operator*=(int x){
		vector<int> res;
		int carry = 0,n=num.size();
		for (int i = 0; i < n||carry; i++) {
			if(i<n){
				carry+=num[i]*x;
			}
			res.push_back(carry%10);
			carry/=10;
		}
		num=res;
		return;
	}
	void operator/=(int x){
		vector<int> res;
		int remainder = 0;
		for (int i = num.size() - 1; i >= 0; i--) {
			remainder = remainder * 10 + num[i];
			res.push_back(remainder / x);
			remainder %= x;
		}
		reverse(res.begin(), res.end());
		while (res.size() > 1 && res.back() == 0) res.pop_back();
		num=res;
		return;
	}
	bool operator<(const BigInt &a){
		int n=num.size(),m=a.num.size();
		if(n!=m){
			return n<m;
		}
		for(int i=n-1;i>=0;i--){
			if(num[i]!=a.num[i]){
				return num[i]<a.num[i];
			}
		}
		return false;
	}
};
