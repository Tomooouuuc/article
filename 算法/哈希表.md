#include<iostream>

#include<algorithm>

#include<chrono>
#include<cstring>
using namespace std;

struct Node{
	char *s;
	Node *next;
};

struct HashTable{
	Node *data;
	int cnt,size;
};

Node* getNewNode(const char *s){
	Node *p=(Node*)malloc(sizeof(Node));
	p->s=strdup(s);
	p->next=NULL;
	return p;
}

HashTable* getNewHashTable(int n){
	HashTable *h=(HashTable*)malloc(sizeof(HashTable));
	h->data=(Node*)malloc(sizeof(Node)*n);
	h->size=n;
	h->cnt=0;
	return h;
}

void clearNode(Node *p){
	if(p==NULL){
		return;
	}
	if(p->s){
		free(p->s);
	}
	free(p);
	return;
}

void clearHashTable(HashTable *h){
	if(h==NULL){
		return;
	}
	for(int i=0;i<h->size;i++){
		Node *p=h->data[i].next,*q;
		while(p){
			q=p->next;
			clearNode(p);
			p=q;
		}
	}
	free(h->data);
	free(h);
	return;
}
